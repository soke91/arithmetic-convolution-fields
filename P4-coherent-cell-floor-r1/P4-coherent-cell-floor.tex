\title{Cell-mean fluctuations in an arithmetic convolution field:\\
exact Rademacher-null variances and a permutation control}
\author{Youngrok Lee\\
\small Independent researcher, Republic of Korea}
\date{}

\begin{document}
\maketitle

\begin{abstract}
Let
\[
 C(N)=\sum_{n<N}\Lambda(n)\mu(N-n),\qquad
 Z(N)=C(N)/\sqrt{V(N)},
\]
where $V(N)$ is the conditional variance obtained by replacing the
nonzero values of $\mu$ by independent Rademacher signs.  We partition
a band of even $N$ according to how many of $3,5,7,11,13$ divide $N$.
For every cell contrast, its Rademacher-null variance is an exactly
computable, doubly centred quadratic form.  Its standard deviation does
not follow the usual inverse-square-root cell-count scale: over eight
octaves, at the three shallowest cells, it shrinks by a factor about
$1.21$, whereas count scaling predicts $11.3$.  In contrast, averaging
the sign-null variance over uniformly random same-size cells gives the
finite-population formula exactly.  In one band, the arithmetic-cell
floors exceed this permutation-averaged scale by factors from $3.8$ to
$105$.  A finite residue-class quantity derived from the pair singular
series captures the floor's profile across cells.  We prove that this
quantity converges to a scale-free constant and compute its main term in
exact rational arithmetic.  Computations for a second partition and a
sweep over nineteen cell shapes provide further consistency checks.  No
limiting law for $Z$, decay rate for the observed contrasts, or Goldbach
consequence is asserted.
\end{abstract}

\section{Introduction}

For even $N$, define
\begin{equation}\label{eq:field}
 C(N)=\sum_{n<N}\Lambda(n)\mu(N-n),\qquad
 V(N)=\sum_{v<N}\mu^2(v)\Lambda(N-v)^2,
 \qquad Z(N)=\frac{C(N)}{\sqrt{V(N)}}.
\end{equation}
The field is related to a reduction used by Huang and Li in their work
on binary Goldbach \cite[Thm.~1, Cor.~1]{HL}; that connection supplies
motivation only.  Nothing below bounds $C(N)$ or yields a Goldbach
consequence.

Fix a band $a$ of even integers and partition it into cells indexed by
the \emph{depth} $d$, the number of primes in
\[
 \{3,5,7,11,13\}
\]
that divide $N$.  This choice is exploratory; computations for eighteen
other prime sets are reported in Measurement~\ref{meas:sweep}.  If
$m_c$ is the mean of $Z$ over a cell $c$ and $\overline m$ is its mean
over the band, our first question is the appropriate reference scale
for $m_c-\overline m$.

The common scale $\operatorname{sd}(Z)/\sqrt{n_c}$, where $n_c$ is the
cell size, treats the observations as self-averaging.  That assumption
is inappropriate here because every $Z(N)$ reads the same underlying
sequence $\mu(v)$; this is the usual design-effect issue for correlated
observations \cite{Kish}.  Under the conditional null that preserves the
support of $\mu^2$ and replaces its nonzero signs independently, the
complete covariance matrix is exactly computable.  Consequently the
variance of every cell contrast is also exactly computable.

The paper has three main conclusions.  First, the cell floor is a
three-term, doubly centred quadratic form and decreases much more slowly
than cell-count scaling predicts.  Second, the permutation average of
the variance for a uniformly random same-size cell is exactly the usual
finite-population expression.  Thus the large floor is a property of
the correspondence between the divisibility cells and the arithmetic
kernel, not of cell size alone.  Third, a residue-class main term for
the pair singular series describes the profile of this floor and is
asymptotically scale-free.

All standardised contrasts below use the conditional Rademacher null.
This supplies an exact variance, not a Gaussian approximation or a
calibrated tail law.  The null preserves the observed squarefree support
of $\mu$ but discards its multiplicative dependence.  Thus the reported
$z_c$ values quantify departure from this specific sign model and are
not calibrated probabilities for the M\"obius function itself.  The
numerical associations involving the observed field are descriptive,
and the selected prime family is not treated as a preregistered choice.

\section{The exact Rademacher-null floor}\label{sec:floor}

We first record the elementary identity behind the calculation.

\begin{lemma}[Contrast variance]\label{lem:contrast}
Let $X=(X_N)_{N\in a}$ be a square-integrable random vector indexed by
a finite set $a$ of size $n$, with covariance kernel
$\Sigma_{NN'}$.  For $c\subseteq a$ of size $n_c\geq1$, put
\[
 m_c=\frac1{n_c}\sum_{N\in c}X_N,\qquad
 \overline m=\frac1n\sum_{N\in a}X_N,\qquad
 Q_{cd}=\sum_{N\in c}\sum_{N'\in d}\Sigma_{NN'}.
\]
Then
\begin{equation}\label{eq:contrast}
 \operatorname{Var}(m_c-\overline m)
 =\frac{Q_{cc}}{n_c^2}-\frac{2Q_{ca}}{n_cn}+\frac{Q_{aa}}{n^2}.
\end{equation}
\end{lemma}

\begin{proof}
Write $m_c-\overline m=w^{\top}X$, where
$w_N=n_c^{-1}\mathbf 1_{N\in c}-n^{-1}$.  Hence
\[
\begin{aligned}
 \operatorname{Var}(m_c-\overline m)
 &=w^{\top}\Sigma w\\
 &=\sum_{N,N'\in a}
   \left(\frac{\mathbf1_{N\in c}}{n_c}-\frac1n\right)
   \Sigma_{NN'}
   \left(\frac{\mathbf1_{N'\in c}}{n_c}-\frac1n\right)\\
 &=\frac{Q_{cc}}{n_c^2}
   -\frac{Q_{ca}+Q_{ac}}{n_cn}
   +\frac{Q_{aa}}{n^2}.
\end{aligned}
\]
Since every covariance matrix is symmetric, $Q_{ac}=Q_{ca}$, and the
last display is exactly \eqref{eq:contrast}.
\end{proof}

Under the sign null, write
$\mu(v)=\mu^2(v)\varepsilon(v)$, where the $\varepsilon(v)$ on the
squarefree support are independent Rademacher variables.

\begin{lemma}[Exact cell moments]\label{lem:cellmom}
Let $a$ be a band of $n$ even integers and $c\subseteq a$ a cell of
size $n_c$.  Set
\[
 u_c(v)=\sum_{N\in c}\frac{\Lambda(N-v)}{\sqrt{V(N)}},
 \qquad
 Q_{cd}=\sum_{1\leq v<N_{\max}}\mu^2(v)u_c(v)u_d(v),
\]
where $\Lambda(k)=0$ for $k\leq0$.  Under the conditional Rademacher
null,
\begin{equation}\label{eq:cellmom}
 \operatorname{Var}_{\varepsilon}(m_c-\overline m)
 =\frac{Q_{cc}}{n_c^2}-\frac{2Q_{ca}}{n_cn}+\frac{Q_{aa}}{n^2}.
\end{equation}
The identity is exact; its reported numerical evaluations use
floating-point arithmetic.
\end{lemma}

\begin{proof}
The null numerator is
\[
 C_{\varepsilon}(N)=\sum_v\mu^2(v)\varepsilon(v)\Lambda(N-v).
\]
Since $E[\varepsilon(v)\varepsilon(v')]=\delta_{vv'}$ and
$\mu^4=\mu^2$,
\[
 \operatorname{Cov}_{\varepsilon}(Z(N),Z(N'))
 =\frac{1}{\sqrt{V(N)V(N')}}
   \sum_v\mu^2(v)\Lambda(N-v)\Lambda(N'-v).
\]
Therefore, for any two cells $c,d$,
\[
\begin{aligned}
 \operatorname{Cov}_{\varepsilon}(n_cm_c,n_dm_d)
 &=\sum_{N\in c}\sum_{N'\in d}
   \operatorname{Cov}_{\varepsilon}(Z(N),Z(N'))\\
 &=\sum_v\mu^2(v)
   \left(\sum_{N\in c}\frac{\Lambda(N-v)}{\sqrt{V(N)}}\right)
   \left(\sum_{N'\in d}\frac{\Lambda(N'-v)}{\sqrt{V(N')}}\right)\\
 &=\sum_v\mu^2(v)u_c(v)u_d(v)=Q_{cd}.
\end{aligned}
\]
Substitution of these three covariance sums into
Lemma~\ref{lem:contrast} gives \eqref{eq:cellmom}.
\end{proof}

Define the null floor and standardised contrast by
\[
 \operatorname{se}_c
 :=\sqrt{\operatorname{Var}_{\varepsilon}(m_c-\overline m)},
 \qquad
 z_c:=\frac{m_c-\overline m}{\operatorname{se}_c}.
\]
All three terms of \eqref{eq:cellmom} are essential.  The approximation
$u_c(v)\approx n_c/\sqrt V$ gives the same leading contribution to each
term, which cancels as $S-2S+S$.  The surviving quantity measures the
departure of the cell kernel from proportionality with the band kernel.

\begin{measurement}[Closed form versus simulation]\label{meas:mc}
In the band $(10^5,2\cdot10^5]$, a simulation with $2000$ independent
sign draws gave
\[
\begin{array}{r|cccccc}
 \text{depth} &0&1&2&3&4&5\\
 n_c&19181&21097&8183&1421&115&3\\\hline
 \text{closed form}&0.140080&0.052724&0.167579&0.273630&0.378073&0.639330\\
 \text{MC}/\text{closed}&0.9920&0.9993&0.9918&1.0044&1.0157&1.0320
\end{array}
\]
The simulation is only a numerical check on the closed form.  Its
largest relative discrepancy is $3.2\%$, at the cell containing three
points.
\end{measurement}

\section{Slow decay and the failure of count scaling}\label{sec:decay}

On average over the $v$ carrying weight, about $n_c/\log N$ terms in
$u_c(v)$ are nonzero, each of typical size $\log N/\sqrt V$.  This gives
the heuristic
\begin{equation}\label{eq:coherent}
 \frac{Q_{cc}}{n_c^2}
 \approx \frac{\sum_v\mu^2(v)}{V(N)}
 \asymp \frac1{\log N},
\end{equation}
using $V(N)\asymp N\log N$; see Appendix~\ref{sec:normaliser}.
The heuristic is an averaged statement and is false at a fixed $v$,
where parity and local congruence obstructions matter.  It predicts the
form of the decay, not its constant or its profile across cells.

\begin{measurement}[Decay across eight octaves]\label{meas:decay}
Fitting $\operatorname{se}_c\propto N^{-b}$ across the eight octaves
from $(6.25\cdot10^4,1.25\cdot10^5]$ to
$(8\cdot10^6,1.6\cdot10^7]$ gives
\[
 b=0.039388,\ 0.039671,\ 0.039451
\]
at depths $0,1,2$.  These values are close to the logarithmic benchmark
$0.0363$ over the same finite range and far from the count benchmark
$1/2$.  Across the factor $128$ in the octave midpoints, the fitted
floor therefore shrinks by about $1.21$, whereas $n_c^{-1/2}$ predicts
$\sqrt{128}=11.3$.

The approximation does not describe every depth uniformly.  The fitted
exponent at depth $5$ is $0.107306$, where the cell contains only $266$
points in the top octave.  At that octave the exact floors from depths
$0$ to $5$ are
\[
 0.1182, 0.0445, 0.1414, 0.2305, 0.3169, 0.4115.
\]
They exceed $\operatorname{sd}(Z)/\sqrt{n_c}$ by factors approximately
$158,62,124,84,33,7$, respectively.  Thus cell count does not describe
either the scale dependence or the cross-cell profile.
\end{measurement}

For reference, the observed standardised contrasts satisfy
\[
 \max_c|z_c|=
 10.099,10.954,11.179,12.968,11.885,11.026,10.019,9.064
\]
over the eight octaves.  At the top octave the six depths give
\[
\begin{array}{r|rrrrrr}
 \text{depth}&0&1&2&3&4&5\\\hline
 z_c&0.04&0.69&-0.06&-1.55&-4.46&-9.06
\end{array}
\]
These are standardised descriptions; no Gaussian or
multiple-comparison calibration is attached to them.

\section{Permutation control}\label{sec:permutation}

Let $K(N,N')$ denote the covariance kernel in the proof of
Lemma~\ref{lem:cellmom}.  Its diagonal is exactly one because
\[
 K(N,N)=\frac{\sum_v\mu^2(v)\Lambda(N-v)^2}{V(N)}=1.
\]
This normalisation makes the average floor of a random cell explicit.
The relabelling below is conditional on the observed field and cell
sizes, as in standard randomisation inference \cite{LR}.

\begin{lemma}[Permutation-averaged floor]\label{lem:permfloor}
Let $c^\pi$ be a uniformly random $n_c$-subset of a band $a$ of size
$n$.  Then
\begin{equation}\label{eq:permfloor}
 \tau_c^2
 :=E_\pi\operatorname{Var}_{\varepsilon}(m_{c^\pi}-\overline m)
 =\frac{n-n_c}{n_c(n-1)}
   \left(1-\frac{Q_{aa}}{n^2}\right).
\end{equation}
For the observed field held fixed,
\begin{equation}\label{eq:permnum}
 \operatorname{Var}_\pi(m_{c^\pi}-\overline m)
 =\frac{n-n_c}{n_c(n-1)}s^2,\qquad
 s^2=\frac1n\sum_N(Z(N)-\overline m)^2,
\end{equation}
and $E_\varepsilon s^2=1-Q_{aa}/n^2$.
\end{lemma}

\begin{proof}
Put $J_c(N)=\mathbf1_c(N)-n_c/n$.  Then
\[
 \operatorname{Var}_{\varepsilon}(m_c-\overline m)
 =\frac1{n_c^2}\sum_{N,N'}J_c(N)J_c(N')K(N,N').
\]
For $p=n_c/n$ and a uniform fixed-size subset,
\[
 E_\pi[J_{c^\pi}(N)J_{c^\pi}(N')]
 =p(1-p)\left(\delta_{NN'}-
 \frac{1-\delta_{NN'}}{n-1}\right).
\]
Using $\sum_NK(N,N)=n$ and
$\sum_{N\ne N'}K(N,N')=Q_{aa}-n$, we obtain
\[
\begin{aligned}
 E_\pi\operatorname{Var}_{\varepsilon}(m_{c^\pi}-\overline m)
 &=\frac{p(1-p)}{n_c^2}
   \left(n-\frac{Q_{aa}-n}{n-1}\right)\\
 &=\frac{p(1-p)}{n_c^2}\frac{n^2-Q_{aa}}{n-1}\\
 &=\frac{n-n_c}{n_c(n-1)}
   \left(1-\frac{Q_{aa}}{n^2}\right),
\end{aligned}
\]
which is \eqref{eq:permfloor}.

For the observed field, put $x_N=Z(N)-\overline m$.  Then
\[
 \sum_Nx_N^2=ns^2,\qquad
 \sum_{N\ne N'}x_Nx_{N'}
 =\left(\sum_Nx_N\right)^2-\sum_Nx_N^2=-ns^2.
\]
Repeating the preceding calculation with
$K(N,N')=x_Nx_{N'}$ gives
\[
 \operatorname{Var}_\pi(m_{c^\pi}-\overline m)
 =\frac{n-n_c}{n_c(n-1)}s^2,
\]
which is \eqref{eq:permnum}.  Finally,
\[
\begin{aligned}
 E_\varepsilon s^2
 &=\frac1n\sum_NE_\varepsilon Z(N)^2
   -E_\varepsilon(\overline m)^2\\
 &=1-\frac{Q_{aa}}{n^2},
\end{aligned}
\]
because $E_\varepsilon Z(N)^2=K(N,N)=1$ and
$E_\varepsilon(\overline m)^2=Q_{aa}/n^2$.
\end{proof}

Thus the count-based scale is recovered exactly, up to the usual
finite-population correction \cite{Cochran}, after averaging the
variance over random same-size cells.  It is not recovered for the
divisibility cells.

\begin{measurement}[Arithmetic and permuted cells]\label{meas:placebo}
In the octave $(2\cdot10^6,4\cdot10^6]$, the true labelling gives
\[
\begin{array}{r|cccccc}
 \text{depth}&0&1&2&3&4&5\\
 n_c&383617&421978&163568&28507&2263&67\\\hline
 z_c&0.1069&1.1133&-0.2314&-2.3990&-5.9997&-11.0258
\end{array}
\]
Ten uniformly sampled fixed-count relabellings gave maxima
\[
 0.9359,1.6073,1.9921,1.5687,2.1392,
 1.1216,1.7178,1.1348,3.2040,1.3192.
\]
This small sample provides descriptive rather than precise permutation
inference: the smallest attainable Monte Carlo $p$-value is $1/11$.

The exact comparison is between the arithmetic floor and
\eqref{eq:permfloor}:
\[
\begin{array}{r|cccccc}
 \text{depth}&0&1&2&3&4&5\\\hline
 \operatorname{se}_c/\tau_c&104.80&42.68&70.28&44.37&16.97&3.83
\end{array}
\]
The permutation average depends on a cell only through its size, whereas
the arithmetic floor depends on which integers carry the label.  The
contrast therefore isolates the correspondence between the cells and
the arithmetic covariance kernel.
\end{measurement}

For the fixed observed numerator, \eqref{eq:permnum} also gives an exact
variance without sampling.  At the same octave,
\[
\begin{array}{r|cccccc}
 \text{depth}&0&1&2&3&4&5\\\hline
 |m_c-\overline m|/\sigma_c&11.25&47.71&16.33&106.89&102.22&42.42
\end{array}
\]
where $\sigma_c^2$ denotes the right-hand side of
\eqref{eq:permnum}.  Chebyshev's inequality bounds the two-sided
permutation probability for the depth-$5$ numerator by
$5.56\cdot10^{-4}$ and the union of the six cellwise events by
$1.28\cdot10^{-2}$.  These bounds concern the numerator, not the ratio
$z_c$, whose denominator changes under relabelling.

\section{A singular-series descriptor}\label{sec:descriptor}

For nonzero $h$, let $\SS_2(h)$ be the Hardy--Littlewood singular series
for a prime pair separated by $h$.  Define ordered off-diagonal pair
averages by
\[
 E_{\mathrm{same},c}[\SS_2]
 =\frac1{n_c(n_c-1)}
   \sum_{\substack{N,N'\in c\\N\ne N'}}\SS_2(N-N'),
\]
\[
 E_{\mathrm{all}}[\SS_2]
 =\frac1{n(n-1)}
   \sum_{\substack{N,N'\in a\\N\ne N'}}\SS_2(N-N'),
\]
and
\begin{equation}\label{eq:Dc}
 D_c=E_{\mathrm{same},c}[\SS_2]-E_{\mathrm{all}}[\SS_2].
\end{equation}
The exact three-term analogue of the centred covariance form is
\[
 B_c=E_{\mathrm{same},c}[\SS_2]
     -2E_{c,a}[\SS_2]+E_{\mathrm{all}}[\SS_2],
\]
where
\[
 E_{c,a}[\SS_2]=\frac1{n_c(n-1)}
 \sum_{N\in c}\sum_{\substack{N'\in a\\N'\ne N}}\SS_2(N-N').
\]
The two-term $D_c$ is used as a descriptor, not asserted to equal the
variance kernel.

\begin{measurement}[Cross-cell profile]\label{meas:Dc}
At the octave $(2\cdot10^6,4\cdot10^6]$, a sample of $400000$ pairs per
cell gives the estimator $\widehatD_c$:
\[
\begin{array}{r|cccccc}
 \text{depth}&0&1&2&3&4&5\\\hline
 \widehatD_c&0.302138&0.046766&0.412504&1.052129&1.946109&3.171298\\
 \operatorname{se}_c&0.12400&0.04662&0.14834&0.24178&0.33253&0.43685
\end{array}
\]
The correlation across the six cells is $0.9805$.  This association is
post hoc and descriptive.  Its useful feature is not the correlation
alone but the scale comparison: $\operatorname{se}_c^2/\widehatD_c$
ranges from $0.04648$ to $0.06018$, while $\widehatD_c$ itself varies by
a factor about $68$.  After matching diagonal and centring conventions
term by term, the ratio has a full width $19.8\%$ of its mean.
\end{measurement}

We next show that $D_c$ has a scale-free residue-class main term.

\begin{lemma}[Progression sum against an envelope]\label{lem:envtv}
Let $w:\mathbb Z\to\mathbb R$ have finite support and total variation
$\operatorname{TV}(w)=\sum_h|w(h+1)-w(h)|$.  For every $q\geq1$ and
$a\in\mathbb Z$,
\[
 \sum_{h\equiv a\ (q)}w(h)
 =\frac1q\sum_hw(h)+O(\operatorname{TV}(w)),
\]
with an absolute implied constant.
\end{lemma}

\begin{proof}
Partition $\mathbb Z$ into consecutive blocks of length $q$, each with
one representative $h_j$ in the class $a\pmod q$.  If $I_j$ is the
$j$th block, then
\[
\begin{aligned}
 \sum_{h\equiv a\,(q)}w(h)-\frac1q\sum_hw(h)
 &=\sum_j\left(w(h_j)-\frac1q\sum_{h\in I_j}w(h)\right)\\
 &=\sum_j\frac1q\sum_{h\in I_j}\bigl(w(h_j)-w(h)\bigr).
\end{aligned}
\]
Let $V_j$ be the variation of $w$ within $I_j$.  For every
$h\in I_j$, $|w(h_j)-w(h)|\leq V_j$, and hence the absolute value of
the last display is at most
\[
 \sum_jV_j\leq\sum_h|w(h+1)-w(h)|=\operatorname{TV}(w).
\]
This proves the stated estimate with implied constant $1$.
\end{proof}

\begin{proposition}[Scale-free main term]\label{prop:scaleinv}
Let
$Q=3\cdot5\cdot7\cdot11\cdot13$, let $a$ be the even integers in an
interval of length $B$, and put $\ell=B/(2Q)$.  Let $R_c$ be a nonempty
set of $m$ even residue classes modulo $2Q$, and let $c$ be their union
inside the band.  Define
\[
 A(h)=\prod_{\substack{2<p\mid h\\p\leq13}}
       \left(1+\frac1{p-2}\right),
 \qquad
 C_2=\prod_{p>2}\frac{p(p-2)}{(p-1)^2},
\]
\[
 \kappa=2C_2\prod_{p>13}\left(1+\frac1{p(p-2)}\right),
\]
and
\[
 \mathcal A_c=\frac1{m^2}\sum_{r,r'\in R_c}A(r-r'),
 \qquad
 \mathcal A_a=\frac1{Q^2}\sum_{r,r'}A(r-r'),
\]
where the second sum is over all even residue classes modulo $2Q$.
Then, for every fixed $\eta\in(0,1)$,
\begin{equation}\label{eq:scaleinv}
 D_c=\kappa(\mathcal A_c-\mathcal A_a)
 +O(\ell^{-1})+O_\eta(B^\eta/\ell)+O_\eta(B^{\eta-1}).
\end{equation}
The constants are uniform over the finitely many depth cells and over
the position of the interval.  Consequently
\[
 D_c\longrightarrow\kappa(\mathcal A_c-\mathcal A_a).
\]
If this limit is nonzero, the one-step exponent
\[
 e_c(B)=-\frac{\log|D_c(2B)/D_c(B)|}{\log2}
\]
tends to zero.
\end{proposition}

\begin{proof}
For even nonzero $h$ write
\[
 \SS_2(h)=2C_2A(h)T(h),\qquad
 T(h)=\prod_{\substack{p\mid h\\p>13}}
       \left(1+\frac1{p-2}\right).
\]
Expanding the second product gives
\[
 T(h)=\sum_{d\mid h}g(d),\qquad
 g(d)=\prod_{p\mid d}\frac1{p-2},
\]
where $d$ is squarefree and all its prime factors exceed $13$.
Every such $d$ is coprime to $2Q$.

Write
\[
 f_c(h)=\#\bigl(R_c\cap(R_c+h)\bigr),
 \qquad
 W_c(h)=\#\{N\in c:N-h\in c\}.
\]
For each residue class counted by $f_c(h)$, the admissible $N$ fill an
interval of length $B-|h|$.  Hence
\begin{equation}\label{eq:pair-envelope}
 W_c(h)=f_c(h)\frac{B-|h|}{2Q}+\rho_c(h),
 \qquad |\rho_c(h)|\leq f_c(h)\leq m.
\end{equation}
Also $n_c=m\ell+O(m)$, so
$n_c(n_c-1)=m^2\ell^2(1+O(\ell^{-1}))$.  The first term in
\eqref{eq:pair-envelope}, after normalisation, is a mixture of triangular
envelopes indexed by pairs of residue classes.  Each such envelope has
total variation $O(B^{-1})$.  Since $(d,2Q)=1$, the Chinese remainder
theorem and Lemma~\ref{lem:envtv} give
\[
 P(d\mid h)=\frac1d+O(B^{-1})
\]
uniformly for $d\leq B$ within each admissible residue-class pair.  The
replacement of the exact class counts by $m\ell$ contributes
$O(\ell^{-1})$.  Summing the per-$d$ discrepancy against $g(d)$
contributes
\[
 O\!\left(\frac1\ell\sum_{d\leq B}g(d)\right)
 =O_\eta(B^\eta/\ell).
\]

The divisor errors and the tail are controlled by Rankin's trick:
\[
\begin{aligned}
 \sum_{d\leq B}g(d)
 &\leq B^\eta
   \prod_{p>13}\left(1+\frac1{(p-2)p^\eta}\right)
   =O_\eta(B^\eta),\\
 \sum_{d>B}\frac{g(d)}d
 &\leq B^{\eta-1}
   \prod_{p>13}\left(1+\frac1{(p-2)p^\eta}\right)
   =O_\eta(B^{\eta-1}).
\end{aligned}
\]
It remains to control the rounding term in
\eqref{eq:pair-envelope}.  Since $T(h)\leq\tau(|h|)$,
\[
 \sum_{0<|h|<B}|\rho_c(h)|T(h)
 \leq m\sum_{0<|h|<B}\tau(|h|)
 =O_\eta(mB^{1+\eta}).
\]
After division by $n_c(n_c-1)$ this is
$O_\eta(B^{\eta-1})$, uniformly over the finitely many cells because
$Q$ is fixed and $1\leq m\leq Q$.  Thus the primes above $13$
contribute, uniformly over each residue-class pair,
\[
 \prod_{p>13}\left(1+\frac1{p(p-2)}\right)
\]
up to the errors in \eqref{eq:scaleinv}.

Finally, $A(h)$ is constant on each pair of residue classes modulo
$2Q$.  Averaging those pairs over the cell and over the whole band gives
$\mathcal A_c$ and $\mathcal A_a$, respectively, up to
$O(\ell^{-1})$.  Consequently
\[
 E_{\mathrm{same},c}[\SS_2]
 =\kappa\mathcal A_c
  +O(\ell^{-1})+O_\eta(B^\eta/\ell)
  +O_\eta(B^{\eta-1}),
\]
and the identical calculation for the full band gives
\[
 E_{\mathrm{all}}[\SS_2]
 =\kappa\mathcal A_a
  +O(\ell^{-1})+O_\eta(B^\eta/\ell)
  +O_\eta(B^{\eta-1}).
\]
Subtracting these two identities yields \eqref{eq:scaleinv}.  Since all
three error terms tend to zero for fixed $\eta\in(0,1)$, a nonzero
limiting constant gives
\[
 \frac{D_c(2B)}{D_c(B)}\longrightarrow1,
 \qquad
 e_c(B)=-\frac{\log|D_c(2B)/D_c(B)|}{\log2}\longrightarrow0.
\]
\end{proof}

\section{Finite computations}\label{sec:computations}

Because $A(h)$ depends only on the residue of $h$ modulo $Q$, the main
term in Proposition~\ref{prop:scaleinv} is a finite rational sum.
Moreover,
\begin{equation}\label{eq:kappa}
 \kappa
 =2\prod_{p\in\{3,5,7,11,13\}}\frac{p(p-2)}{(p-1)^2}
 =\frac{11011}{8192}.
\end{equation}

\begin{measurement}[Residue main term]\label{meas:maincoef}
The exact rational values of $\mathcal A_c-\mathcal A_a$ at depths
$0$ through $5$ are
\[
 \frac{635653709}{2854051200},\quad
 \frac{10743253}{313945632},\quad
 \frac{318004704}{1037775739},\quad
 \frac{296100736}{378194817},\quad
 \frac{13823072}{9546537},\quad
 \frac{78976}{33033}.
\]
All are nonzero.  Multiplying by \eqref{eq:kappa} gives
\[
\begin{array}{r|cccccc}
 \text{depth}&0&1&2&3&4&5\\\hline
 \text{main term}&0.299361&0.045996&0.411876&1.052351&1.946236&3.213542\\
 \widehatD_c&0.302138&0.046766&0.412504&1.052129&1.946109&3.171298\\
 \text{residual/MCSE}&2.0&0.6&0.5&0.2&0.1&260.8
\end{array}
\]
Here MCSE is the pair-sampling standard error, not the null floor
$\operatorname{se}_c$.  Four depths lie within two MCSEs; depth $0$ is
at the displayed boundary.  At depth $5$, the cell is a single residue
class and contains $67$ points.  Every shift is $2Qk$ with
$0<|k|<67$, so the missing tail of the divisor expansion is explicitly
computable.  Its contribution is $0.043558$, compared with the observed
residual $0.042244$.

For the full residue set, the same Euler-factor calculation gives
$\kappa\mathcal A_a=2$; thus the limiting whole-band average of
$\SS_2$ over even shifts is $2$, consistently with the classical
average of the singular series \cite{Gallagher,Pintz}.
Writing
\[
 D'_c=E_{\mathrm{same},c}[\SS_2]-2
\]
removes the common sampling error in the estimated band average.  With
the depth-$5$ finite-divisor correction included, the comparison becomes
\[
\begin{array}{r|cccccc}
 \text{depth}&0&1&2&3&4&5\\\hline
 D'_c&0.300893&0.045521&0.411259&1.050884&1.944864&3.170053\\
 \text{prediction}&0.299361&0.045996&0.411876&1.052351&1.946236&3.169984\\
 \text{residual/MCSE}&-1.13&0.36&0.45&1.07&1.16&-0.43
\end{array}
\]
All entries lie within $1.2$ pair-sampling standard errors.  This
finite-band refinement was obtained post hoc; it is an explicit
calculation rather than a fitted correction.
\end{measurement}

\begin{measurement}[A second cell shape]\label{meas:secondcell}
For the depth partition generated by $S=(3,5,7)$, the modulus is
$Q'=105$ and
\[
 K=2\prod_{p\in S}\frac{p(p-2)}{(p-1)^2}=\frac{175}{128}.
\]
The pair averages were evaluated exhaustively by autocorrelation, so the
row labelled exact has no pair-sampling error:
\[
\begin{array}{r|cccc}
 \text{depth}&0&1&2&3\\\hline
 m&48&44&12&1\\
 \text{main term}&0.295736&0.126485&0.825521&2.375000\\

At the three multiclass depths the exact value lies between the residue

algebraic consistency check.  The exact difference from the main term
decomposes into a residue-weight error, a finite-divisor deficit, and a
covariance between the low-prime factor $A_S$ and the remaining Euler

\end{measurement}

\begin{measurement}[Nineteen cell shapes]\label{meas:sweep}
The same computation was applied to every pair and triple from
$\{3,5,7,11,13\}$ that the band resolves, four quadruples, the full
five-prime set, and that set with $17$ adjoined.  Among $70$ cells with
$n_c\geq1000$, every residue main term differs from the exhaustive value
by less than $0.01$ and by less than the magnitude of the exact value;
the largest absolute difference is $2.9766\cdot10^{-3}$.  Eight smaller
cells were reported but not scored.  All $70$ scored differences have
the same sign: the main term exceeds the exact value.  This is empirical
consistency across cell shapes, not a proof of the sign.
\end{measurement}

Exhaustive autocorrelation also separates the deterministic $D_c$ from
its sampled estimator.  Over three consecutive octaves the relative
spreads of the six exact $D_c$ values are
\[
 0.0001, 0.0004, 0.0002, 0.0002, 0.0022, 0.0128,
\]
and the corresponding fitted exponents are
\[
 -0.000059, -0.000285, -0.000136, -0.000164,
 -0.001584, -0.009202.
\]
These finite computations agree with the scale-free conclusion of
Proposition~\ref{prop:scaleinv}; they do not supply a convergence rate.

\section{Conclusion}

The conditional Rademacher null makes the covariance of the convolution
field explicit.  The variance of a cell contrast is therefore the exact
three-term form \eqref{eq:cellmom}, rather than a count-based standard
error.  For the divisibility cells studied here the resulting floor is
large and slowly varying, while its average over uniformly random
same-size cells reduces exactly to the finite-population formula.

The excess pair singular series $D_c$ describes much of the floor's
cross-cell profile.  Its residue-class main term is rational and
asymptotically scale-free, and finite computations show the same
mechanism across additional cell shapes.  The observed association
between $D_c$ and the floor remains descriptive: no equality between the
two kernels is proved, the cell family was selected exploratorily, and
no limiting distribution or decay law for the observed contrasts is
claimed.

\section{The normalising scale}\label{sec:normaliser}

\begin{proposition}\label{prop:V}
Let
\[
 W(N)=\sum_{w<N}\Lambda(w)^2,\qquad
 \AAA(N)=\prod_{q\nmid N}\left(1-\frac1{q(q-1)}\right).
\]
Then
\[
 V(N)=W(N)\AAA(N)(1+o(1))
 \sim \AAA(N)N\log N.
\]
\end{proposition}

\begin{proof}
Since $\Lambda(w)^2$ is supported on prime powers,
\[
 V(N)=\sum_{p<N}(\log p)^2\mu^2(N-p)
      +O(\sqrt N\log^2N).
\]
For a prime $q\nmid N$, the forbidden unit class
$p\equiv N\pmod{q^2}$ contributes the local factor
$1-1/\varphi(q^2)=1-1/(q(q-1))$.  If $q\mid N$, the condition
$q^2\mid N-p$ forces the exceptional prime $p=q$, giving local density
$1$.  Mirsky's theorem on squarefree shifted primes \cite{Mir49}
therefore supplies the factor $\AAA(N)$.  This product is bounded below
by the positive product over all primes, uniformly in $N$.
More explicitly, if
\[
 R(N)=\sum_{p<N}\mu^2(N-p),
\]
then Mirsky's theorem gives
\[
 R(N)=\AAA(N)\operatorname{Li}(N)+o\!\left(\frac N{\log N}\right).
\]

To insert the weight, take
$\varepsilon(N)=1/\log\log N$.  The contribution of the primes below
$N^{1-\varepsilon}$ is bounded by
\[
 (\log N)^2\pi(N^{1-\varepsilon})=o(N\log N).
\]
On the remaining range,
$(\log p)^2=(1+o(1))(\log N)^2$, uniformly in $p$.  Hence
\[
\begin{aligned}
 V(N)
 &=(1+o(1))(\log N)^2R(N)+o(N\log N)\\
 &=(1+o(1))\AAA(N)N\log N.
\end{aligned}
\]
Finally, partial summation from $\theta(x)\sim x$ gives
$W(N)\sim N\log N$.  Dividing the last display by this asymptotic
yields $V(N)=W(N)\AAA(N)(1+o(1))$.
\end{proof}
